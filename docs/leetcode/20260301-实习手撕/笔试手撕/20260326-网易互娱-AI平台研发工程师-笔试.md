// 妖伞传递 - 事件驱动模拟
package main

import (
    "bufio"
    "fmt"
    "os"
)

func simulate(n int, start, end []int64) []int64 {
    res := make([]int64, n)

    // 0 1 2 3 ，一共四个人，也可以认为是 start 和 end 数组的索引（是一样的）
    queue := make([]int, 0, n)

    // 这个front和next实际上是queue的索引，对应队首/队尾指针
    // front++: 用于出队
    // next++: 用于入队
    front := 0
    next := 1

    curPos := start[0]
    queue = append(queue, 0)  // 第1个人持伞出发，入队

    for front < len(queue) || next < n {

        // 1. 离队
        // 终点已在当前位置左侧的队首立即离队
        for front < len(queue) && end[queue[front]] <= curPos {
            front++
        }

        // 只要队列空了，就走这里
        if front == len(queue) {
            if next >= n {
                break
            }
            // 队空时：start 最小的未加入者去取伞
            queue = append(queue, next)
            next++
            continue
        }

        // 2. 计算nextStart nextEnd
        head := queue[front]

        // nextStart: 下一个未加入者出生点
        // nextEnd: 队首元素的终点（准备离开队伍的节点，的终点）
        var nextStart int64
        if next < n {
            // 事件1 新的人出现
            nextStart = start[next]
        } else {
            nextStart = 1 << 62
        }
        // 事件2 当前持伞的人到终点
        nextEnd := end[head]

        // 3. 通过比较 nextStart 和 nextEnd 去计算 event位置，取最小值, 并更新 curPos
        // 取较近的事件点，累加持伞者贡献
        // 初始: event = nextStart
        // 情况1:
        // ------event--------------------------
        // ------nextStart-----nextEnd----------
        // ------event--------------------------
        // 情况2:
        // --------------------event------------
        // ------nextEnd-------nextStart--------
        // ------event--------------------------
        event := min(nextStart, nextEnd)
        res[head] += event - curPos

        curPos = event

        // 4. 入队
        // 将沿途出生点 <= curPos 的人依次加入队列
        for next < n && start[next] <= curPos {
            queue = append(queue, next)
            next++
        }
    }

    return res
}

func min(a, b int64) int64 {
    if a < b {
        return a
    }

    return b
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    
    var n int
    fmt.Fscan(reader, &n)
    start := make([]int64, n)
    end := make([]int64, n)

    for i := 0; i < n; i++ {
        fmt.Fscan(reader, &start[i], &end[i])
    }

    // 1. 出队，如果队列为空，就重新入队把 next 入队
    // 2. 寻找 nextStart 和 nextEnd
    // 3. 寻找 event，计算 res[next]，更新 curPos 为 event
    // 4. 入队
    res := simulate(n, start, end)

    for i := 0; i < n; i++ {
        fmt.Print(res[i])
        if i < n-1 {
            fmt.Print(" ")
        }
    }
}
