## 暴力解法

```
输入：
23121
2 4 9

DFS会枚举：
2
22
222
...
22999 ✅（最大合法）
24xxx ❌（剪枝）

输出：
22999
```



```go
​```
时间复杂度为 O(n^k)
n 为数组 nums 长度
k 为 target 的位数
​```

package main

import (
    "fmt"
    "bufio"
    "os"
    "strings"
    "strconv"
)

func findMax(nums []int, target int) int {
    res := -1
    
    targetStr := strconv.Itoa(target)
    maxLen := len(targetStr)
    
    var dfs func(cur, length int)
    dfs = func(cur, length int) {
        // 如果当前值超出了 target，就直接返回退出
        if cur >= target {
            return
        }
        
        // 更新 res
        if cur > res {
            res = cur
        }
        
        // 如果当前长度和 target 位数相同，则不继续增加位数
        if length == maxLen {
            return
        }
        
        // 枚举更长位数的
        for i := 0; i < len(nums); i++ {
            dfs(cur*10 + nums[i], length+1)
        }
    }
    
    // 遍历每个数字
    for i := 0; i < len(targetStr); i++ {
        dfs(targetStr[i], 1)
    }
    
    if res == target {
        return -1
    }
    
    return res
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    
    line1, _ := reader.ReadString('\n')
    line2, _ := reader.ReadString('\n')

    desc := strings.Fields(line1)
    numsStr := strings.Fields(line2)

    target, _ := strconv.Atoi(desc[0])

    nums := make([]int, 0)
    for _, v := range numsStr {
        val, _ := strconv.Atoi(v)
        nums = append(nums, val)
    }

    res := findMax(nums, target)

    fmt.Println(res)
}
