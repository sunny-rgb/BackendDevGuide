## 题目1：链表排序（按照字典序）

```go
package main

import (
    "fmt"
    "os"
    "sort"
)

func less(a, b []int) bool {
    minLen := len(a)
    if len(b) < minLen {
        minLen = len(b)
    }

    for i := 0; i < minLen; i++ {
        if a[i] != b[i] {
            return a[i] < b[i]
        }
    }

    return len(a) < len(b)
}

func main() {
    var n int
    fmt.Fscan(os.Stdin, &n)

    arrs := make([][]int, n)

    for i := 0; i < n; i++ {
        var k int
        fmt.Fscan(os.Stdin, &k)

        arrs[i] = make([]int, k)

        for j := 0; j < k; j++ {
            fmt.Fscan(os.Stdin, &arrs[i][j])
        }
    }

    sort.Slice(arrs, func(i, j int) bool {
        return less(arrs[i], arrs[j])
    })

    for _, arr := range arrs {
        for i, v := range arr {
            if i > 0 {
                fmt.Print(" ")
            }
            fmt.Print(v)
        }
        fmt.Println()
    }
}
```

package main

import (
    "fmt"
    "os"
)

    var temp int
    if start[0] >= birthday[0] {
        temp = end[0] - start[0] - 1
        if !compare(start, birthday) {
            temp++
        }

        if compare(end, birthday) {
            temp++
        }
    } else {
        temp = end[0] - birthday[0]
        if compare(end, birthday) {
            temp++

    }

    return temp
}

func isLeap(day []int) bool {
    if day[0] % 400 == 0 || day[0] % 4 == 0 && day[0] % 100 != 0{
        return true
    }

    return false
}

// 判断是否已经过过生日
func compare(day, birthday []int) bool {

    birthMonth := birthday[1]
    birthDay := birthday[2]


    // 2月29生日特殊处理
    if birthMonth == 2 && birthDay == 29 {

        // 当前年份不是闰年
        if !isLeap(day) {
            birthDay = 28
        }
    }


    if day[1] > birthMonth {
        return true
    }

    if day[1] < birthMonth {
        return false
    }

    return day[2] >= birthDay
}

func main() {
    var n int
    fmt.Fscan(os.Stdin, &n)

    for i := 0; i < n; i++ {
        birthday := make([]int, 3)
        start := make([]int, 3)
        end := make([]int, 3)

        for j := 0; j < 3; j++ {
            fmt.Fscan(os.Stdin, &birthday[j])
        }

        for j := 0; j < 3; j++ {
            fmt.Fscan(os.Stdin, &start[j])
        }

        for j := 0; j < 3; j++ {
            fmt.Fscan(os.Stdin, &end[j])
        }

        res := calculate(birthday, start, end)

    }
}
