## 题目1：阶乘平方数

```go
package main

import (
    "fmt"
    "os"
    "math"
)

// 计算阶乘
func calculate_x(x int64) int64 {
    if x <= 1 {
        return 1
    }

    return x * calculate_x(x-1)
}

// 判断是否是完全平方数
func check(x int64) bool {
    root := int64(math.Sqrt(float64(x)))

    return root*root == x
}

func calculate(n int64) []int64 {
    res := make([]int64, 0)

    for i := int64(1); i < n; i ++ {
        value := calculate_x(i) + 1

        // 判断条件 1
        if value > n {
            break
        }

        // 判断条件 2
        if check(value) {
            res = append(res, i)
        }
    }

    if len(res) == 0 {
        return []int64{-1}
    }

    return res
}

func main() {
    var nums int

    fmt.Fscan(os.Stdin, &nums)

    for i := 0; i < nums; i++ {
        var n int64
        fmt.Fscan(os.Stdin, &n)

        res := calculate(n)

        for j := 0; j < len(res); j++ {
            fmt.Print(res[j])
            if j != len(res) - 1 {
                fmt.Print(" ")
            }
        }

        fmt.Println()
    }
}

```



## 题目2：线性组合计数

### 解法1

```go
package main

import (
    "fmt"
    "os"
    "bufio"
    "strings"
    "strconv"
)

func gcd(x, y int64) int64 {
    for y != 0 {
        x, y = y, x % y
    }

    return x
}

func calculate_n(x, y, l, r int64) int64 {
    g := gcd(x, y)
    return r/g - (l-1)/g
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    numsStr, _ := reader.ReadString('\n')
    numsStr = strings.Fields(numsStr)[0]
    nums, _ := strconv.Atoi(numsStr)

    for i := 0; i < nums; i++ {
        line_i, _ := reader.ReadString('\n')
        parts := strings.Fields(line_i)
        
        x, _ := strconv.Atoi(parts[0])
        y, _ := strconv.Atoi(parts[1])
        l, _ := strconv.Atoi(parts[2])
        r, _ := strconv.Atoi(parts[3])

        n_i := calculate_n(int64(x), int64(y), int64(l), int64(r))

        fmt.Println(n_i)
    }
}
```
