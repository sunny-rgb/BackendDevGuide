### 3.1 Self-Attention

实现 Scaled Dot-Product Attention 算法。

```python
import numpy as np

def softmax(x):
    exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
    return exp_x / np.sum(exp_x, axis=-1, keepdims=True)

def scaled_dot_product_attention(L, d, X, Wq, Wk, Wv):
    # 计算 Q、K、V
    Q = np.dot(X, Wq)
    K = np.dot(X, Wk)
    V = np.dot(X, Wv)

    # 计算注意力分数
    scores = np.dot(Q, K.T) / np.sqrt(d)

    # 对每一行做 softmax，得到注意力权重
    attention_weights  = softmax(scores)

    # 加权求和
    output = np.dot(attention_weights, V)

    return output


if __name__ == "__main__":
    L, d = map(int, input().split())
    X = np.array([list(map(float, input().split())) for _ in range(L)])
    Wq = np.array([list(map(float, input().split())) for _ in range(d)])
    Wk = np.array([list(map(float, input().split())) for _ in range(d)])
    Wv = np.array([list(map(float, input().split())) for _ in range(d)])

    output = scaled_dot_product_attention(L, d, X, Wq, Wk, Wv)
    
    for row in output:
        print(' '.join(f'{val:.4f}' for val in row))
```

### 3.2 Multi-Head Attention

```python
import numpy as np

def softmax(x):
    """数值稳定的 softmax"""
    exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
    return exp_x / np.sum(exp_x, axis=-1, keepdims=True)

def scaled_dot_product_attention(Q, K, V):
    """
    单头注意力机制
    Q, K, V: (L, d_k)
    返回: (L, d_k)
    """
    d_k = Q.shape[-1]
    scores = np.dot(Q, K.T) / np.sqrt(d_k)  # (L, L)
    weights = softmax(scores)                # (L, L)
    output = np.dot(weights, V)              # (L, d_k)
    return output

def multi_head_attention(X, Wq, Wk, Wv, Wo, h):
    """
    Multi-Head Attention
    
    参数:
        X: (L, d) 输入序列
        Wq, Wk, Wv, Wo: (d, d) 投影矩阵
        h: 头数
    
    返回:
        output: (L, d)
    """
    L, d = X.shape
    d_k = d // h  # 每个头的维度
    
    # 1. 线性投影得到 Q, K, V
    Q = np.dot(X, Wq)  # (L, d)
    K = np.dot(X, Wk)  # (L, d)
    V = np.dot(X, Wv)  # (L, d)
    
    # 2. 拆分成多头: (L, h, d_k)
    Q = Q.reshape(L, h, d_k)
    K = K.reshape(L, h, d_k)
    V = V.reshape(L, h, d_k)
    
    # 3. 每个头独立计算注意力
    heads = []
    for i in range(h):
        Qi = Q[:, i, :]  # (L, d_k)
        Ki = K[:, i, :]  # (L, d_k)
        Vi = V[:, i, :]  # (L, d_k)
        head_i = scaled_dot_product_attention(Qi, Ki, Vi)  # (L, d_k)
        heads.append(head_i)
    
    # 4. 拼接所有头: (L, h*d_k) = (L, d)
    concat = np.concatenate(heads, axis=1)  # (L, d)
    
    # 5. 最终输出投影
    output = np.dot(concat, Wo)  # (L, d)
    
    return output


if __name__ == "__main__":
    L, d, h = map(int, input().split())
    
    X = np.array([list(map(float, input().split())) for _ in range(L)])
    
    Wq = np.array([list(map(float, input().split())) for _ in range(d)])
    Wk = np.array([list(map(float, input().split())) for _ in range(d)])
    Wv = np.array([list(map(float, input().split())) for _ in range(d)])
    Wo = np.array([list(map(float, input().split())) for _ in range(d)])
    
    output = multi_head_attention(X, Wq, Wk, Wv, Wo, h)
    

    for row in output:
        print(' '.join(f'{val:.4f}' for val in row))
```
